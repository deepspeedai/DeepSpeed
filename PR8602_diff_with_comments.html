231+// ─── Dual-weight GEMV (b=1 decode) ───
232+// Reads gate.weight and up.weight directly — no concat, no weight copies.
233+// Each warp handles one output feature: lanes cooperatively compute two
234+// dot products (gate row and up row share the same hidden vector read),
235+// then applies silu(a)*b. Memory-bound at b=1; coalesced reads achieve
236+// near-peak HBM bandwidth with zero tiling or tensor-core complexity.

238+__global__ void dual_gemv_silu_mul_kernel(const __nv_bfloat16* __restrict__ hidden,
239+ const __nv_bfloat16* __restrict__ gate_w,
240+ const __nv_bfloat16* __restrict__ up_w,
241+ __nv_bfloat16* __restrict__ out,
242+ int out_features,
243+ int in_features)
244+{
245+ int warp_id = (blockIdx.x * blockDim.x + threadIdx.x) >> 5;
246+ int lane = threadIdx.x & 31;
247+ if (warp_id >= out_features) return;
248+
249+ const __nv_bfloat16* gate_row = gate_w + (long)warp_id * in_features;
250+ const __nv_bfloat16* up_row = up_w + (long)warp_id * in_features;
251+
252+ // Each lane accumulates every 32nd element (coalesced across lanes).
253+ float gate_acc = 0.0f, up_acc = 0.0f;
254+ for (int i = lane; i < in_features; i += 32) {
255+ float h = __bfloat162float(hidden[i]);
256+ gate_acc += h * __bfloat162float(gate_row[i]);
257+ up_acc += h * __bfloat162float(up_row[i]);
258+ }
259+
260+ // Warp-level reduction for both dot products simultaneously.
261+ for (int offset = 16; offset > 0; offset >>= 1) {
262+ gate_acc += __shfl_down_sync(0xffffffff, gate_acc, offset);
263+ up_acc += __shfl_down_sync(0xffffffff, up_acc, offset);
264+ }
265+
266+ if (lane == 0) {
267+ float activated = SILU(gate_acc);
268+ out[warp_id] = __float2bfloat16_rn(activated * up_acc);
269+ }

377+// ─── Custom b=1 decode attention (graph-compatible, no mask) ───
378+// Streams softmax(q @ K[0:pos]^T / sqrt(d)) @ V[0:pos] with GQA. The valid
379+// KV length comes from a GPU-resident write_pos tensor, so the kernel is
380+// CUDA-graph replayable (address fixed, only the value changes per step).
381+// One block per KV head; its 8 warps split as (query head, KV chunk):
382+// warp w handles query head w % q_per_kv over KV positions strided by
383+// split = 8 / q_per_kv, then per-head partial online-softmax states are
384+// merged through shared memory. Each lane owns a contiguous 16B slice of
385+// the head dimension (HEAD_DIM/32 elements) for coalesced K/V row loads.
386+
387+template <int HEAD_DIM>
388+__global__ void __launch_bounds__(512) decode_attn_kernel(const __nv_bfloat16* __restrict__ q,
389+ const __nv_bfloat16* __restrict__ K,
390+ const __nv_bfloat16* __restrict__ V,
391+ const int64_t* __restrict__ write_pos,
392+ __nv_bfloat16* __restrict__ out,
393+ int num_q_heads,
394+ int num_kv_heads,
395+ int max_len,
396+ float scale)
397+{
398+ constexpr int EPL = HEAD_DIM / 32; // head-dim elements owned per lane
399+ const int q_per_kv = num_q_heads / num_kv_heads;
400+ const int warps_per_block = blockDim.x >> 5;
401+ const int split = warps_per_block; // KV chunks per query head
402+
403+ // One block per query head: all the block's warps split its KV range,
404+ // giving enough independent streams to cover HBM latency at b=1.
405+ const int q_head = blockIdx.x;
406+ const int kv_head = q_head / q_per_kv;
407+ const int warp_id = threadIdx.x >> 5;
408+ const int lane = threadIdx.x & 31;
409+ const int s = warp_id;
410+
411+ const int pos = (int)(*write_pos); // valid KV positions: 0..pos inclusive
412+
413+ const __nv_bfloat16* Kg = K + (long)kv_head * max_len * HEAD_DIM;
414+ const __nv_bfloat16* Vg = V + (long)kv_head * max_len * HEAD_DIM;
415+
416+ float q_reg[EPL];
417+#pragma unroll
418+ for (int e = 0; e < EPL; e++)
419+ q_reg[e] = __bfloat162float(q[(long)q_head * HEAD_DIM + lane * EPL + e]);
420+
421+ float max_score = -INFINITY;
422+ float sum_exp = 0.0f;
423+ float out_reg[EPL];
424+#pragma unroll
425+ for (int e = 0; e < EPL; e++) out_reg[e] = 0.0f;
426+
427+ // UNROLL positions in flight so their K loads overlap and hide HBM latency.
428+ constexpr int UNROLL = 4;
429+ for (int base = s; base <= pos; base += UNROLL * split) {
430+ float k_reg[UNROLL][EPL];
431+ int idx[UNROLL];
432+#pragma unroll
433+ for (int u = 0; u < UNROLL; u++) {
434+ idx[u] = base + u * split;
435+ if (idx[u] <= pos) {
436+ const __nv_bfloat16* krow = Kg + (long)idx[u] * HEAD_DIM;
437+#pragma unroll
438+ for (int e = 0; e < EPL; e++) k_reg[u][e] = __bfloat162float(krow[lane * EPL + e]);
439+ }
440+ }
441+#pragma unroll
442+ for (int u = 0; u < UNROLL; u++) {
443+ if (idx[u] > pos) continue; // uniform across the warp
444+ float dot = 0.0f;
445+#pragma unroll
446+ for (int e = 0; e < EPL; e++) dot += q_reg[e] * k_reg[u][e];
447+#pragma unroll
448+ for (int off = 16; off > 0; off >>= 1) dot += __shfl_down_sync(0xffffffff, dot, off);
449+ float score = __shfl_sync(0xffffffff, dot, 0) * scale;
450+
451+ float new_max = fmaxf(max_score, score);
452+ float rescale = expf(max_score - new_max); // 0 when max_score is still -inf
453+ float weight = expf(score - new_max);
454+ sum_exp = sum_exp * rescale + weight;
455+#pragma unroll
456+ for (int e = 0; e < EPL; e++) out_reg[e] *= rescale;
457+ const __nv_bfloat16* vrow = Vg + (long)idx[u] * HEAD_DIM;
458+#pragma unroll
459+ for (int e = 0; e < EPL; e++)
460+ out_reg[e] += weight * __bfloat162float(vrow[lane * EPL + e]);
461+ max_score = new_max;
462+ }
463+ }
464+
465+ // Merge the per-chunk partial softmax states (max, sum, weighted V) in
466+ // shared memory; an empty chunk contributes 0 via exp(-inf - M).
467+ __shared__ float s_max[16];
468+ __shared__ float s_sum[16];
469+ __shared__ float s_out[16][HEAD_DIM];
470+
471+ s_max[warp_id] = max_score;
472+ s_sum[warp_id] = sum_exp;
473+#pragma unroll
474+ for (int e = 0; e < EPL; e++) s_out[warp_id][lane * EPL + e] = out_reg[e];
475+ __syncthreads();
476+
477+ if (s == 0) { // the first chunk's warp combines all chunks for its head
478+ float M = -INFINITY;
479+ for (int t = 0; t < split; t++) M = fmaxf(M, s_max[t]);
480+ float S = 0.0f;
481+ float acc[EPL];
482+#pragma unroll
483+ for (int e = 0; e < EPL; e++) acc[e] = 0.0f;
484+ for (int t = 0; t < split; t++) {
485+ float wm = expf(s_max[t] - M);
486+ S += s_sum[t] * wm;
487+#pragma unroll
488+ for (int e = 0; e < EPL; e++) acc[e] += s_out[t][lane * EPL + e] * wm;
489+ }
490+ float inv = 1.0f / S;
491+#pragma unroll
492+ for (int e = 0; e < EPL; e++)
493+ out[(long)q_head * HEAD_DIM + lane * EPL + e] = __float2bfloat16_rn(acc[e] * inv);
494+ }

574+// ─── Fused residual add + RMSNorm (b=1 decode) ───
575+// One kernel replaces the two PyTorch ops per norm site (hidden += residual,
576+// then RMSNorm(hidden) * weight). hidden is updated in-place to hold the new
577+// residual stream; the normalized output is written to a separate tensor that
578+// feeds the next sublayer. One thread block covers the whole row so the
579+// sum-of-squares reduction stays block-local.
580+__global__ void fused_add_norm_kernel(__nv_bfloat16* __restrict__ hidden,
581+ const __nv_bfloat16* __restrict__ residual,
582+ const __nv_bfloat16* __restrict__ weight,
583+ __nv_bfloat16* __restrict__ out,
584+ float eps,
585+ int dim)
586+{
587+ extern __shared__ float s_data[];
588+ __shared__ float s_reduce[32];
589+
590+ int tid = threadIdx.x;
591+ int n_threads = blockDim.x;
592+
593+ float local_ss = 0.0f;
594+ for (int i = tid; i < dim; i += n_threads) {
595+ float h = __bfloat162float(hidden[i]) + __bfloat162float(residual[i]);
596+ s_data[i] = h;
597+ hidden[i] = __float2bfloat16_rn(h);
598+ local_ss += h * h;
599+ }
600+
601+ // Warp then cross-warp reduction of the sum of squares.
602+ for (int offset = 16; offset > 0; offset >>= 1)
603+ local_ss += __shfl_down_sync(0xffffffff, local_ss, offset);
604+ int warp_id = tid >> 5;
605+ if ((tid & 31) == 0) s_reduce[warp_id] = local_ss;
606+ __syncthreads();
607+ if (warp_id == 0) {
608+ float v = (tid < (n_threads >> 5)) ? s_reduce[tid] : 0.0f;
609+ for (int offset = 16; offset > 0; offset >>= 1)
610+ v += __shfl_down_sync(0xffffffff, v, offset);
611+ if (tid == 0) s_reduce[0] = rsqrtf(v / dim + eps);
612+ }
613+ __syncthreads();
614+ float rms = s_reduce[0];
615+
616+ for (int i = tid; i < dim; i += n_threads)
617+ out[i] = __float2bfloat16_rn(s_data[i] * rms * __bfloat162float(weight[i]));
618+}
